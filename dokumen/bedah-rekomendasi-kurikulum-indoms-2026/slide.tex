\begin{frame}{Profil Lulusan (\textit{Qualification Profile})}

\begin{itemize}
\item \textbf{Profil lulusan (\textit{qualification profile})} adalah gambaran tentang peran yang dapat dijalankan lulusan setelah menyelesaikan suatu program studi, beserta kompetensi utama yang diperlukan untuk menjalankan peran tersebut dalam dunia kerja maupun masyarakat.
\item Profil lulusan menjadi dasar penyusunan \textbf{capaian pembelajaran lulusan (CPL)} dan kurikulum.
\end{itemize}

\begin{itemize}
\item Profil lulusan disusun berdasarkan masukan dari dosen, alumni dan pengguna lulusan
\item Apakah profil lulusan sudah ditampilkan di website?
\end{itemize}

\end{frame}

\begin{frame}{Tujuan Pendidikan Program Studi (\textit{Program Educational Objectives})}

\textbf{Tujuan Pendidikan Program Studi (\textit{Program Educational Objectives}/PEO)} adalah pernyataan tentang pencapaian profesional dan pengembangan diri yang diharapkan diraih alumni beberapa tahun setelah lulus, berdasarkan visi program studi dan kebutuhan pemangku kepentingan.

\begin{block}{Contoh}
Alumni mampu mengembangkan karier dengan menerapkan matematika dan komputasi untuk menyelesaikan masalah secara profesional, beretika, serta terus meningkatkan kompetensinya
\end{block}

\textbf{TPP/PEO} berfokus pada pencapaian alumni setelah beberapa tahun lulus, sedangkan \textbf{CPL/PLO} berfokus pada kemampuan yang harus dimiliki saat lulus.

\end{frame}

\begin{frame}{Tujuan Pendidikan Program Studi (\textit{Program Educational Objectives})}

\begin{itemize}
\item Apakah program studi Matematika UINSA sudah menyusun tujuan pendidikan program studi?
\item Tujuan pendidikan program studi disusun berdasarkan profil lulusan dan masukan dari stakeholder (mahasiswa, dosen, alumni dan pengguna alumni)
\end{itemize}

Hubungan antara profil lulusan, tujuan pendidikan program studi dan capaian pembelajaran lulusan
\begin{itemize}
\item \textbf{Profil lulusan}: peran yang dipersiapkan bagi lulusan
\item \textbf{Tujuan pendidikan program studi}: pencapaian yang diharapkan diraih alumni beberapa tahun setelah lulus dalam menjalankan peran tersebut
\item \textbf{Capaian pembelajaran lulusan}: kemampuan yang harus dimiliki saat lulus untuk mendukung pencapaian PEO
\end{itemize}

%Misalnya, profil lulusan \textbf{Peneliti Muda} dapat mendasari PEO: ``\textit{Alumni berkembang dalam kegiatan penelitian dengan menerapkan matematika dan komputasi, bekerja sama dalam tim, serta menjunjung integritas ilmiah}.''

\end{frame}

\begin{frame}{Tujuan Pendidikan Program Studi berdasarkan rekomendasi kurikulum minimal IndoMS}

%Tujuan pendidikan program studi berdasarkan rekomendasi kurikulum minimal IndoMS
\begin{enumerate}
\item \textbf{Karier dan penerapan keilmuan}: mengembangkan karier di bidang pendidikan, penelitian, konsultasi data, industri, jasa, pemerintahan, atau kewirausahaan melalui penerapan matematika, statistika, dan komputasi untuk menyelesaikan permasalahan dan mendukung pengambilan keputusan berbasis bukti
\item \textbf{Etika dan tanggung jawab profesional}: menjalankan peran profesional dengan menjunjung integritas, etika akademik dan profesi, serta tanggung jawab terhadap penggunaan data, kualitas hasil pekerjaan, dan dampaknya bagi masyarakat
\item \textbf{Kolaborasi dan inovasi}: berkontribusi dalam tim lintas disiplin melalui komunikasi yang efektif, pengelolaan kegiatan secara bertanggung jawab, dan pengembangan solusi atau inovasi yang relevan dengan kebutuhan pemangku kepentingan
\item \textbf{Pengembangan diri berkelanjutan}: mengembangkan keahlian dan beradaptasi terhadap perkembangan ilmu pengetahuan dan teknologi melalui studi lanjut, pelatihan profesional, penelitian, atau pembelajaran mandiri sepanjang hayat
\end{enumerate}

\end{frame}
