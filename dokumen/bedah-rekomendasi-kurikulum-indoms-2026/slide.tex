\begin{frame}{Profil Lulusan (\textit{Qualification Profile})}

\begin{itemize}
\item \textbf{Profil lulusan (\textit{qualification profile})} adalah gambaran tentang peran yang dapat dijalankan lulusan setelah menyelesaikan suatu program studi, beserta kompetensi utama yang diperlukan untuk menjalankan peran tersebut dalam dunia kerja maupun masyarakat.
\item Profil lulusan menjadi dasar penyusunan \textbf{capaian pembelajaran lulusan (CPL)} dan kurikulum.
\end{itemize}

\begin{frame}{Tujuan Pendidikan Program Studi (\textit{Program Educational Objectives})}

\textbf{Tujuan Pendidikan Program Studi (\textit{Program Educational Objectives}/PEO)} adalah pernyataan tentang pencapaian profesional dan pengembangan diri yang diharapkan diraih alumni beberapa tahun setelah lulus, berdasarkan visi program studi dan kebutuhan pemangku kepentingan.

\begin{block}{Contoh}
Alumni mampu mengembangkan karier dengan menerapkan matematika dan komputasi untuk menyelesaikan masalah secara profesional, beretika, serta terus meningkatkan kompetensinya
\end{block}

\textbf{TPP/PEO} berfokus pada pencapaian alumni setelah beberapa tahun lulus, sedangkan \textbf{CPL/PLO} berfokus pada kemampuan yang harus dimiliki saat lulus.

\end{frame}

\begin{frame}{Tujuan Pendidikan Program Studi (\textit{Program Educational Objectives})}

\begin{itemize}
\item Apakah program studi Matematika UINSA sudah menyusun tujuan pendidikan program studi?
\item Tujuan pendidikan program studi disusun berdasarkan profil lulusan dan masukan dari stakeholder (mahasiswa, dosen, alumni dan pengguna alumni)
\item Apakah tujuan pendidikan program studi sudah ditampilkan di website?
\end{itemize}

\end{frame}

\begin{frame}{Tujuan Pendidikan Program Studi (\textit{Program Educational Objectives})}

\begin{block}{Hubungan antara profil lulusan, tujuan pendidikan program studi dan capaian pembelajaran lulusan}
\begin{itemize}
\item \textbf{Profil lulusan}: peran yang dipersiapkan bagi lulusan
\item \textbf{Tujuan pendidikan program studi}: pencapaian yang diharapkan diraih alumni beberapa tahun setelah lulus dalam menjalankan peran tersebut
\item \textbf{Capaian pembelajaran lulusan}: kemampuan yang harus dimiliki saat lulus untuk mendukung pencapaian PEO
\end{itemize}
\end{block}

\begin{itemize}
\item Misalnya, profil lulusan \textbf{Peneliti Muda} dapat mendasari PEO: ``\textit{Alumni berkembang dalam kegiatan penelitian dengan menerapkan matematika dan komputasi, bekerja sama dalam tim, serta menjunjung integritas ilmiah}.''
\item Hubungan profil lulusan dan tujuan pendidikan prodi dituliskan dalam tabel
\end{itemize}

\end{frame}

\begin{frame}{Tujuan Pendidikan Program Studi berdasarkan rekomendasi kurikulum minimal IndoMS}

\begin{enumerate}
\item \textbf{Karier dan penerapan keilmuan}: mengembangkan karier di bidang pendidikan, penelitian, konsultasi data, industri, jasa, pemerintahan, atau kewirausahaan melalui penerapan matematika, statistika, dan komputasi untuk menyelesaikan permasalahan dan mendukung pengambilan keputusan berbasis bukti
\item \textbf{Etika dan tanggung jawab profesional}: menjalankan peran profesional dengan menjunjung integritas, etika akademik dan profesi, serta tanggung jawab terhadap penggunaan data, kualitas hasil pekerjaan, dan dampaknya bagi masyarakat
\item \textbf{Kolaborasi dan inovasi}: berkontribusi dalam tim lintas disiplin melalui komunikasi yang efektif, pengelolaan kegiatan secara bertanggung jawab, dan pengembangan solusi atau inovasi yang relevan dengan kebutuhan pemangku kepentingan
\item \textbf{Pengembangan diri berkelanjutan}: mengembangkan keahlian dan beradaptasi terhadap perkembangan ilmu pengetahuan dan teknologi melalui studi lanjut, pelatihan profesional, penelitian, atau pembelajaran mandiri sepanjang hayat
\end{enumerate}

\end{frame}

\begin{frame}[allowframebreaks]{Capaian pembelajaran lulusan (\textit{program learning outcomes}) kurikulum Matematika UINSA}

\begin{enumerate}
\item Mampu mengintegrasikan nilai-nilai Pancasila dan semangat kebangsaan dalam praktik kehidupan bermasyarakat, berbangsa, dan bernegara, serta menganalisis isu-isu kebangsaan, keagamaan, sosial dan teknologi, secara logis, kritis, dan etis dengan berpijak pada kearifan lokal untuk mendukung pengambilan keputusan akademik yang bertanggung jawab dan berkelanjutan.
\item Mampu mengintegrasikan nilai-nilai Islam dengan ilmu pengetahuan dan teknologi melalui model integrasi keilmuan Twin Towers dalam memahami dan merespons persoalan kehidupan secara kritis, inovatif, dan bertanggung jawab.
\item Mampu menunjukkan sikap bertanggung jawab dalam mengambil keputusan dengan menerapkan, menganalisis, dan mengintegrasikan teori matematika dan terapannya sebagai alternatif pemecahan masalah di berbagai bidang.
\item Mampu berpikir logis, kritis, sistematis, dan inovatif dalam mengevaluasi penyelesaian masalah matematis melalui eksplorasi, abstraksi, generalisasi, dan bukti formal, dengan menjunjung tinggi etika akademik.
\item Mampu menginternalisasi etika akademik dalam memecahkan permasalahan matematika terapan dengan atau tanpa mengoperasikan perangkat lunak.
\item Mampu menyusun deskripsi saintifik secara ilmiah, menghubungkan hasil pekerjaannya dengan konsep matematika yang relevan, serta mengomunikasikan secara lisan dan tulisan secara sistematis, dengan sikap terbuka terhadap pendapat dan temuan orisinal orang lain.
\item Mampu memecahkan masalah matematika atau terapannya dengan kerja sama dengan penuh peduli santun.
\item Mampu menerapkan prinsip belajar sepanjang hayat serta adaptif terhadap perkembangan teori matematika dan terapannya, dengan semangat kemandirian dan kejuangan dalam meningkatkan kompetensi diri.
\item 	Mampu menerapkan model matematis dalam berbagai persoalan di bidang matematika, sains, dan kehidupan sehari-hari, serta menentukan pendekatan solusi yang tepat dalam rangka berkontribusi dalam peningkatan mutu kehidupan bermasyarakat.
\item Mampu berkolaborasi dan beradaptasi dengan dinamika sosial budaya masyarakat, serta menunjukkan sikap terbuka terhadap perbedaan melalui penguasaan konsep-konsep humaniora dan sosial serta penerapan keterampilan komunikasi, dan pengembangan jaringan kerja lintas budaya.
\end{enumerate}

\end{frame}

\begin{frame}[allowframebreaks]{Capaian pembelajaran lulusan (\textit{program learning outcomes}) rekomendasi kurikulum minimal IndoMS}

\begin{enumerate}
\item Memahami konsep serta struktur matematika dasar, aljabar, analisis, dan geometri secara kritis, logis, dan etis, untuk membuktikan dan menggeneralisasi pernyataan matematis, dalam konteks akademik dan profesional, dengan menjunjung tinggi integritas dan ketelitian ilmiah.
\item Menganalisis dan menilai prinsip matematika diskrit dan aljabar dalam merumuskan dan memecahkan masalah kompleks, secara sistematis dan bertanggung jawab, dalam konteks pembelajaran, riset awal, dan pengembangan solusi praktis, dengan sikap disiplin dan terbuka terhadap kritik ilmiah.
\item Memformulasikan model matematika berbasis matematika dasar, aljabar, dan statistika, untuk merepresentasikan dan menyelesaikan fenomena nyata, dalam konteks akademik, industri, atau kebijakan publik, dengan menunjukkan tanggung jawab, objektivitas, dan kesadaran etika.
\item Menginterpretasikan data dan informasi kuantitatif menggunakan statistika dan teknologi kecerdasan buatan, untuk menghasilkan rekomendasi berbasis bukti, dalam konteks penelitian, analisis big data, dan pengambilan keputusan, dengan sikap kritis, profesional, dan menjaga kerahasiaan data.
\item Mengevaluasi asumsi, metode, dan hasil kajian matematika dengan memanfaatkan analisis, geometri, dan statistika, untuk menilai validitas dan keterbatasan solusi, dalam konteks diskursus akademik dan pemecahan masalah nyata, secara reflektif, jujur, dan berorientasi pada kualitas ilmiah.
\item Mengomunikasikan gagasan, model, dan hasil kajian matematika, statistika, dan pemodelan secara argumentatif, sistematis, dan kolaboratif, dalam konteks forum akademik maupun profesional, dengan menjunjung etika komunikasi ilmiah dan tanggung jawab kolektif.
\item Mengkolaborasikan pengetahuan matematika, keterampilan analitik, dan sikap profesional untuk merumuskan solusi inovatif dan mengambil keputusan strategis berbasis big data, dalam konteks kerja tim multidisiplin, riset, industri, atau kewirausahaan, dengan menunjukkan kepemimpinan, adaptivitas, dan pembelajaran sepanjang hayat.
\end{enumerate}

\end{frame}
