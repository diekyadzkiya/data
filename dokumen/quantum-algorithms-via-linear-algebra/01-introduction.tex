\begin{frame}{Are quantum computers possible?}
    \begin{itemize}
		\item Classical computation operates on \textit{bits}, but quantum computation operates on \textit{qubits}
		\item Such question is yet to be resolved
		\item Fortunately, the question does not pose a barrier to us: we plan to study \textbf{quantum algorithms}
		\item The mathematics required to understand them is linear algebra: vectors, matrices, and their basic properties
	\end{itemize}
    
\end{frame}

% 3
\begin{frame}{The model: the space and the states}
    \begin{itemize}
		\item The state is described by a single unit vector $\mathbf{a} = \begin{bmatrix} a_0 & \dots & a_{N-1} \end{bmatrix}^T$ that lies in some fixed vector space of dimension $N = 2^n$, for some $n$, where each entry $a_k$ is a real or complex number
		\item Each entry is called an \textbf{amplitude}
		\item In the \textbf{standard basis}, the basis states are denoted by the vectors $\mathbf{e}_k$
		\item We further identify $k$ with the $k$-th string $x$ in a fixed total ordering of $\{0,1\}^n$
		\item Since basis states correspond to all the length-$n$ binary strings, we have $N = 2^n$
		\item Any vector that is not a basis state is a \textbf{superposition}
		\item The squared absolute value $|a_k|^2$ gives the probability of measuring the system to be in the basis state $\mathbf{e}_k$
		\item Vector $\mathbf{a}$ is a unit vector, namely $\sum_{k=0}^{N-1} |a_k|^2 = 1$
	\end{itemize}
\end{frame}

% 4
\begin{frame}{The operations}
    \begin{itemize}
		\item Every state $\mathbf{a}$ must be a point on the unit sphere
		\item The operations must map any point on the unit sphere to some (other) point on the unit sphere
		\item Unitary operations: linear and onto the whole sphere (invertible)
		\item The operations are represented by matrices, i.e. unitary matrices
		\item Compositions of unitary operations are unitary
		\item Start: the elementary vector $e_0 = \begin{bmatrix} 1 & 0 & 0 & \dots & 0 \end{bmatrix}^T$
		\item Move: apply unitary transformation $U$, i.e. $b = U a$
		\item End: we get information by making a measurement. If the final state is $c$, then $k$ is seen with probability $|c_k|^2$. The output is just the index $k$ without the probability.
	\end{itemize}
\end{frame}
