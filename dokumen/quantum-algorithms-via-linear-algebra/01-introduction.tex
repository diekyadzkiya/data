\begin{frame}{Mengapa dimensi ruang adalah $N=2^n$?}

Label basis adalah semua string biner dengan panjang tetap $n$. Setiap posisi memiliki dua pilihan, sehingga terdapat $2^n$ label.
\[
 n=2:\qquad \{00,01,10,11\},\qquad N=4.
\]
\begin{center}
\begin{tabular}{cccc}\toprule
Indeks $k$ & String biner & Vektor basis & Dimensi vektor\\\midrule
0 & 00 & $e_0=(1,0,0,0)^T$ & 4\\
1 & 01 & $e_1=(0,1,0,0)^T$ & 4\\
2 & 10 & $e_2=(0,0,1,0)^T$ & 4\\
3 & 11 & $e_3=(0,0,0,1)^T$ & 4\\\bottomrule
\end{tabular}
\end{center}
\textbf{Pembedaan penting:} dua qubit berarti vektor keadaan berdimensi empat. Satu hasil pengukuran lengkap adalah satu string yang terdiri atas dua bit.
\bookref{Subbab 1.2, hlm. 3; tabel merupakan contoh pendamping}

\end{frame}

\begin{frame}{Keadaan harus berupa vektor satuan}
\setlength{\abovedisplayskip}{5pt}\setlength{\belowdisplayskip}{5pt}

Vektor keadaan yang sah memenuhi
\[
 a=\begin{pmatrix}a_0\\\vdots\\a_{N-1}\end{pmatrix},\qquad
 \|a\|^2=\sum_{k=0}^{N-1}|a_k|^2=1.
\]
\begin{itemize}
\item Untuk amplitudo real, $|a_k|^2=a_k^2$.
\item Untuk amplitudo kompleks $a_k=x+iy$, berlaku $|a_k|^2=x^2+y^2$.
\item Syarat norma satu membuat jumlah probabilitas hasil sama dengan satu.
\end{itemize}
\begin{block}{Ruang lingkup buku}
Bab ini menggunakan keadaan murni yang direpresentasikan oleh satu vektor. Teori keadaan campuran tidak diperlukan untuk mengikuti alur ini.
\end{block}
\vspace{-4pt}
\bookref{Subbab 1.2, hlm. 3--4}

\end{frame}

\]

\end{column}

\textbf{Vektor kompleks}

\]

\]

Gunakan $|i|^2=1$, bukan $i^2=-1$, untuk menghitung probabilitas.
\end{column}

\end{frame}

\begin{frame}{Keadaan basis dan superposisi}
\setlength{\abovedisplayskip}{5pt}\setlength{\belowdisplayskip}{5pt}

Setiap keadaan memiliki ekspansi
\[
 a=\sum_{k=0}^{N-1}a_ke_k.
\]
\textbf{Keadaan basis:} hanya satu komponen yang tidak nol. Wakil standarnya adalah $e_k$.
\par\medskip
\textbf{Superposisi terhadap basis standar:} sedikitnya dua komponen memiliki amplitudo tidak nol.
\[
 a=\frac35e_0+\frac45e_1
 \quad\text{dan}\quad
 b=\frac1{\sqrt2}e_0+\frac{i}{\sqrt2}e_1.
\]
Satu pengukuran menghasilkan satu label basis. Kita tidak mendapatkan semua komponen vektor sekaligus.
\par\medskip
{\small Catatan ketelitian: kelipatan fase $e^{i\gamma}e_k$ tetap merepresentasikan keadaan basis yang sama, bukan superposisi baru.}
\vspace{-4pt}
\bookref{Subbab 1.2, hlm. 3--4; catatan fase adalah klarifikasi matematis}

\end{frame}

\begin{frame}{1.3 The Operations: perubahan keadaan harus sah}
\small

Operasi sebelum pengukuran ditulis sebagai perkalian matriks:
\[
 b=Ua.
\]
Buku mensyaratkan operasi yang linear dan mempertahankan himpunan vektor satuan. Operasi tersebut bersifat reversibel.
\begin{block}{Syarat matriks uniter}
\[
 U^\dagger U=I,\qquad U^{-1}=U^\dagger.
\]
$U^\dagger$ berarti transpose konjugat. Untuk matriks real, syaratnya $U^TU=I$ dan istilah khususnya adalah matriks ortogonal.
\end{block}
\small
Persamaan ini memperjelas istilah \textit{unitary} yang diperkenalkan Bab 1. Buku memberikan rincian sifat matriks tersebut pada Bab 3.
\bookref{Subbab 1.3, hlm. 5}

\end{frame}

\begin{frame}{Mengapa uniter mempertahankan normalisasi?}

\textcolor{teal}{\small Penjabaran aljabar pendamping}
\[
 b=Ua.
\]
Dengan definisi norma dan sifat transpose konjugat,
\begin{align*}
 \|b\|^2&=b^\dagger b\\
 &=(Ua)^\dagger(Ua)\\
 &=a^\dagger U^\dagger Ua\\
 &=a^\dagger a=\|a\|^2.
\end{align*}
Jadi, jika keadaan awal sah, keadaan setelah operasi uniter juga sah.
\par\medskip
\textbf{Cek sederhana:} $2I$ invertibel, tetapi bukan uniter. Untuk vektor satuan $a$, berlaku $\|2Ia\|=2$, sehingga hasilnya tidak ternormalisasi.
\bookref{pendamping Subbab 1.3, hlm. 5}

\end{frame}

\begin{frame}{Program adalah komposisi matriks}
\small

Jika $U_1$ bekerja lebih dahulu, kemudian $U_2$, maka
\[
 a\xrightarrow{U_1}b=U_1a\xrightarrow{U_2}c=U_2U_1a.
\]
\textbf{Matriks paling kanan bekerja lebih dahulu.}
\par\medskip
Komposisi matriks uniter tetap uniter:
\[
 (U_2U_1)^\dagger(U_2U_1)
 =U_1^\dagger U_2^\dagger U_2U_1
 =U_1^\dagger U_1=I.
\]
Untuk $t$ langkah,
\[
 c=U_tU_{t-1}\cdots U_1e_0.
\]
\begin{block}{Cara membaca algoritma}
Telusuri vektor setelah setiap operasi. Jangan langsung mengubah amplitudo menjadi probabilitas sebelum tahap pengukuran.
\end{block}
\vspace{-4pt}
\bookref{Subbab 1.3, hlm. 5}

\end{frame}

\draw[->,thick] (a)--(b);\draw[->,thick] (b)--(c);
\end{tikzpicture}
\end{center}
\begin{enumerate}
\item \textbf{Start:} mulai dari keadaan basis yang mewakili string $0^n$.
\item \textbf{Move:} gunakan operasi uniter yang diizinkan.
\item \textbf{End:} ukur sehingga indeks $k$ muncul dengan probabilitas $|c_k|^2$.
\end{enumerate}
Keluaran satu percobaan adalah indeks $k$, bukan angka probabilitas $|c_k|^2$ dan bukan seluruh vektor $c$.
\bookref{Subbab 1.3, hlm. 5--6}

\end{frame}

\begin{frame}{Menerima atau menolak: himpunan keluaran $S$}

\[
 \Pr(\text{accept})=\sum_{k\in S}|c_k|^2.
\]

Hasil ukur adalah satu indeks. Aturan klasik kemudian memetakan indeks itu menjadi keputusan.

\begin{frame}{1.4 Where Is the Input?: masukan berada pada operasi}
\small

Jika keadaan awal selalu $e_0$, bagaimana kita menyatakan masalah yang berbeda?
\begin{block}{Jawaban buku}
Masukan dapat dikodekan dalam pilihan transformasi. Untuk masukan $x$, tulis matriks program sebagai $U_x$.
\end{block}
\[
 c_x=U_xe_0.
\]
Sebagian operasi awal dapat menyiapkan masukan, sedangkan operasi berikutnya melakukan pemrosesan yang sama untuk berbagai masukan.
\[
 U_x=V A_x,\qquad c_x=V A_xe_0.
\]
Di sini $A_x$ menyiapkan informasi masukan, dan $V$ memprosesnya. Notasi ini merupakan penjabaran dari gagasan buku.
\bookref{Subbab 1.4, hlm. 6}

\end{frame}

\begin{frame}{Peluang berhasil dan peran pengulangan}

\textcolor{teal}{\small Contoh pendamping probabilitas}
Misalkan sebuah percobaan menghasilkan kandidat benar dengan probabilitas $p=3/4$, dan kandidat benar dapat diverifikasi.
\par\medskip
Untuk $r$ percobaan independen:
\[
 \Pr(\text{semua gagal})=(1-p)^r,
\]
\[
 \Pr(\text{minimal satu berhasil})=1-(1-p)^r.
\]
Untuk $r=3$,
\[
 \Pr(\text{minimal satu berhasil})=1-\left(\frac14\right)^3=\frac{63}{64}.
\]
\small
Ini bukan aturan untuk menerima jika ada satu keluaran \textit{accept} pada setiap masalah keputusan. Cara menggabungkan hasil harus mengikuti jenis masalah dan cara memverifikasi solusi.
\bookref{motivasi pengulangan pada Subbab 1.5, hlm. 7; rumus adalah pengayaan pendamping}

\end{frame}

\begin{frame}{1.6 Summary and Notes: keluaran dapat deterministik}

Jika keadaan akhir adalah vektor basis $c=e_j$, maka
\[
 \Pr(k=j)=1,\qquad \Pr(k\neq j)=0.
\]
\extra
Untuk matriks uniter $U$,
\[
 e_0\xrightarrow{U}Ue_0\xrightarrow{U^\dagger}U^\dagger Ue_0=e_0.
\]
Keadaan antara dapat berupa superposisi, tetapi hasil akhir selalu 0.
\par\medskip
Dengan demikian, model kuantum mencakup keluaran pasti maupun probabilistik. Bab 1 mengajak pembaca menerima probabilitas sebagai bagian penting dalam analisis algoritma.
\bookref{Subbab 1.6, hlm. 8}

\end{frame}

\begin{frame}{Lampiran A: pembahasan latihan 1}

\[
 a=\frac1{\sqrt{10}}(1,2,i,2)^T.
\]
\textbf{1. Normalisasi}
\[
 \|a\|^2=\frac{|1|^2+|2|^2+|i|^2+|2|^2}{10}
 =\frac{1+4+1+4}{10}=1.
\]
\textbf{2. Distribusi pengukuran}
\[
 (p_0,p_1,p_2,p_3)=\left(\frac1{10},\frac4{10},\frac1{10},\frac4{10}\right).
\]
\textbf{3. Peluang menerima}
\[
 S=\{1,3\}\quad\Longrightarrow\quad
 \Pr(\text{accept})=p_1+p_3=\frac8{10}=\frac45.
\]
Indeks 1 dan 3 masing-masing mewakili string 01 dan 11.

\end{frame}

\begin{frame}{Lampiran B: pembahasan latihan 2}

\[
 U^2=\frac1{25}\begin{pmatrix}3&-4\\4&3\end{pmatrix}^2
 =\frac1{25}\begin{pmatrix}-7&-24\\24&-7\end{pmatrix}.
\]
Maka
\[
 U^2e_0=\begin{pmatrix}-7/25\\24/25\end{pmatrix},\qquad
 p_0=\frac{49}{625},\quad p_1=\frac{576}{625}.
\]
Pemeriksaan normalisasi: $49/625+576/625=1$.
\par\medskip
Sebaliknya,
\[
 U^TUe_0=Ie_0=e_0,\qquad (p_0,p_1)=(1,0).
\]
Dua kali rotasi yang sama berbeda dari rotasi yang diikuti inversnya.

\end{frame}

\begin{frame}{Lampiran C: pembahasan latihan 3}

\[
 VA_0e_0=\begin{pmatrix}3/5\\4/5\end{pmatrix},\qquad
 VA_1e_0=\begin{pmatrix}-4/5\\3/5\end{pmatrix}.
\]
\[
 \Pr(k=1\mid x=0)=\frac{16}{25},\qquad
 \Pr(k=1\mid x=1)=\frac9{25}.
\]
\begin{itemize}
\item Kedua peluang tidak nol. Satu hasil 1 tidak menentukan masukan secara pasti.
\item Pengulangan menghasilkan informasi statistik untuk membedakan kedua distribusi.
\item Probabilitas masukan bersyarat pada hasil, $\Pr(x\mid k)$, berbeda dari $\Pr(k\mid x)$ dan memerlukan informasi prior jika hendak dihitung.
\end{itemize}
\textbf{Inti konsep:} memasukkan informasi ke dalam operasi tidak menjamin bahwa satu pengukuran tertentu langsung mengungkap seluruh informasi masukan.

\end{frame}

\begin{frame}{Lampiran D: ketelitian istilah geometri dan pengukuran}

\textbf{Vektor satuan dan keadaan fisik.}\par
Untuk perhitungan, gunakan vektor bernorma satu. Dua vektor yang berbeda hanya oleh fase global, $a$ dan $e^{i\gamma}a$, merepresentasikan keadaan fisik yang sama.
\par\medskip
\textbf{Bola satuan dan ruang proyektif.}\par
Keduanya berkaitan, tetapi tidak identik sebagai himpunan. Ruang keadaan murni diperoleh dengan mengidentifikasi vektor-vektor satuan yang hanya berbeda fase global. Rincian ini tidak diperlukan untuk contoh utama.
\par\medskip
\textbf{Proyeksi dan pengukuran.}\par
Komponen $a_ke_k$ belum tentu bernorma satu. Jika hasil pengukuran lengkap adalah $k$, keadaan bersyarat dinormalisasi kembali dan ekuivalen dengan $e_k$. Panjang komponen sebelum normalisasi menentukan probabilitas.
\par\medskip
\small Catatan ini memperjelas bahasa geometris pada Subbab 1.2--1.3. Tidak perlu diajarkan sebagai teori baru pada pertemuan pertama.

\end{frame}

\begin{frame}{Lampiran E: probabilitas tidak menyimpan tanda amplitudo}
\small

\textcolor{teal}{\small Pengayaan opsional, penghubung ke pembahasan berikutnya}
\[
 a=\frac1{\sqrt2}\begin{pmatrix}1\\1\end{pmatrix},\qquad
 b=\frac1{\sqrt2}\begin{pmatrix}1\\-1\end{pmatrix}.
\]
Keduanya menghasilkan distribusi $(1/2,1/2)$ jika langsung diukur.
Namun, untuk matriks uniter
\[
 W=\frac1{\sqrt2}\begin{pmatrix}1&1\\1&-1\end{pmatrix},
\]
berlaku
\[
 Wa=e_0,\qquad Wb=e_1.
\]
\textbf{Kesimpulan:} distribusi hasil langsung belum menggambarkan seluruh informasi keadaan. Tanda dan fase amplitudo dapat memengaruhi hasil setelah operasi selanjutnya.
\par\medskip
\small Contoh ini merupakan pengayaan, bukan materi eksplisit Bab 1. Tidak diperlukan untuk menyelesaikan latihan utama.

\end{frame}
