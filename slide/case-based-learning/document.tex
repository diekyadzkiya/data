\section{Paradigma \& Urgensi}
% ---------------------------------------------------

% SLIDE 2
\begin{frame}{Mengapa Kita Berada di Sini?}
    \begin{itemize}
        \item \textbf{Tuntutan Kurikulum \& IKU 7:}\\
        Kriteria Indikator Kinerja Utama perguruan tinggi mensyaratkan bobot evaluasi berbasis kasus/proyek $\ge 50\%$.
        \vspace{0.3cm}
        \item \textbf{Tantangan Pembelajaran Matematika:}
        \begin{itemize}
            \item Seringkali didominasi oleh metode ceramah dan derivasi teoretis yang kaku.
            \item Mahasiswa kerap bertanya: \textit{``Untuk apa materi ini di dunia nyata?''}
        \end{itemize}
        \vspace{0.3cm}
        \item \textbf{Solusi:}\\
        Menjembatani teori abstrak dengan fenomena nyata melalui \textbf{\textit{Case-Based Method (CBM)}}.
    \end{itemize}

\vspace{0.3cm}
Apa itu \textit{Case-Based Method} (CBM)?
	
\end{frame}

% SLIDE 3
\begin{frame}{Apa itu \textit{Case-Based Method} (CBM)?}
    \textbf{Definisi:}\\
    Pendekatan pembelajaran aktif di mana mahasiswa belajar melalui analisis, diskusi, dan penyelesaian masalah dari kasus yang kontekstual.
    
    \vspace{0.5cm}
    \textbf{Pergeseran Paradigma:}
    \begin{itemize}
        \item \textbf{Dosen:} Dari \textit{Content Provider} (Penceramah) $\rightarrow$ \textit{Facilitator \& Coach} (Fasilitator/Pembimbing).
        \item \textbf{Mahasiswa:} Dari \textit{Passive Receiver} (Penerima Pasif) $\rightarrow$ \textit{Active Problem Solver} (Pemecah Masalah Aktif).
    \end{itemize}

\vspace{0.5cm}
Studi kasus apa yang ideal di matematika?
	
\end{frame}

% SLIDE 4
\begin{frame}{Anatomy of a ``Good Math Case''}
    Apa yang membuat sebuah studi kasus matematika menjadi ideal?
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Kontekstual \& Realistis:}\\ Mengangkat fenomena nyata dari industri, sains, data riil, atau sistem komputasi.
        \vspace{0.2cm}
        \item \textbf{Terbuka (\textit{Open-ended}):}\\ Tidak selalu memiliki satu jawaban mutlak. Memungkinkan berbagai variasi pemodelan, asumsi, atau strategi penyelesaian.
        \vspace{0.2cm}
        \item \textbf{Mendorong Penalaran Berjenjang:}\\ Menuntut mahasiswa melakukan eksplorasi data, pembentukan model matematis, komputasi, hingga interpretasi hasil.
    \end{itemize}
\end{frame}

% ---------------------------------------------------
\section{Desain \& Integrasi Strategis}
% ---------------------------------------------------

% SLIDE 5
\begin{frame}{Kerangka Kerja 4 Tahap CBM Matematika}
    \begin{block}{Alur Penyelesaian Masalah}
        \centering
        [1. Problem \& Data] $\rightarrow$ [2. Formulation/Model] $\rightarrow$ [3. Solving/Verifying] $\rightarrow$ [4. Interpretation]
    \end{block}
    \vspace{0.3cm}
    \begin{enumerate}
        \item \textbf{Identifikasi Masalah \& Batasan Sistem:}\\ Memahami fenomena fisis/bisnis dan menetapkan parameter/variabel.
        \item \textbf{Formulasi Formal \& Spesifikasi:}\\ Mentransformasikan masalah naratif menjadi model matematika, logika, atau komputasi.
        \item \textbf{Penyelesaian \& Eksekusi Komputasi:}\\ Menggunakan metode analitik, numerik, algoritma, atau \textit{tools} perangkat lunak.
        \item \textbf{Interpretasi \& Evaluasi:}\\ Menerjemahkan hasil matematis kembali ke rekomendasi praktis di domain aplikasi nyata.
    \end{enumerate}
\end{frame}

% Slide 6a: Klaster Matematika Dasar (Kalkulus 1, 2, Peubah Banyak, Vektor)
\begin{frame}{Contoh CBM – Klaster Matematika Dasar}
    \begin{block}{1. Kalkulus 1 \& Kalkulus 2}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Efisiensi kapasitas tampung waduk dan laju erosi tanah.
            \item \textbf{Alur Kasus:} Analisis data laju debit air masuk/keluar $\rightarrow$ Menentukan fungsi Laju \& Luas Penampang $\rightarrow$ Menghitung nilai ekstrem (optimasi) dan akumulasi volume air dengan Integral $\rightarrow$ Menyusun skenario waktu puncak banjir.
        \end{itemize}
    \end{block}

    \begin{block}{2. Kalkulus Peubah Banyak \& Analisis Vektor}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Pemetaan potensi energi angin di wilayah pesisir.
            \item \textbf{Alur Kasus:} Analisis data medan vektor kecepatan angin $\rightarrow$ Menghitung \textit{gradient}, \textit{divergence}, dan \textit{curl} $\rightarrow$ Menggunakan Teorema Stokes/Green untuk fluks energi $\rightarrow$ Menginterpretasikan lokasi optimal kincir angin.
        \end{itemize}
    \end{block}
\end{frame}

% Slide 6b: Klaster Aljabar (Aljabar Linier Elem., Aljabar I \& II, Aljabar Linier)
\begin{frame}{Contoh CBM – Klaster Aljabar}
    \begin{block}{3. Aljabar Linier Elementer \& Aljabar Linier}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Pemeringkatan dokumen/laman web (\textit{PageRank}) atau Model Input-Output Ekonomis Leontief.
            \item \textbf{Alur Kasus:} Analisis keterhubungan antar-sektor/laman $\rightarrow$ Membentuk matriks stokastik \& sistem persamaan linier $\rightarrow$ Menghitung \textit{eigenvalue} \& \textit{eigenvector} dominan $\rightarrow$ Menyusun ranking stabilitas sistem.
        \end{itemize}
    \end{block}

    \begin{block}{4. Aljabar I \& Aljabar II (Struktur Aljabar)}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Keamanan enkripsi pesan digital (Kriptografi RSA / Kurva Eliptik).
            \item \textbf{Alur Kasus:} Membedah struktur Grup/Ring modulo $n$ $\rightarrow$ Menguji sifat koset dan Isomorfisma $\rightarrow$ Menganalisis ketahanan kunci enkripsi berbasis sifat teoretis aljabar.
        \end{itemize}
    \end{block}
\end{frame}

% Slide 6c: Klaster Analisis (Analisis I \& II, Logika Mat., Fungsi Peubah Kompleks)
\begin{frame}{Contoh CBM – Klaster Analisis \& Logika Matematika}
    \begin{block}{5. Logika Matematika \& Matematika Diskrit}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Verifikasi logika sirkuit digital / ekspresi boolean logika perbankan.
            \item \textbf{Alur Kasus:} Identifikasi \textit{requirement} sistem $\rightarrow$ Formulasi proposisi \& inferensi logika $\rightarrow$ Membuktikan validitas penalaran / kebebasan kontradiksi.
        \end{itemize}
    \end{block}

    \begin{block}{6. Analisis I, Analisis II, \& Fungsi Peubah Kompleks}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Analisis kegagalan konvergensi pada algoritma pemrosesan sinyal.
            \item \textbf{Alur Kasus:} Membedah algoritma yang mengalami \textit{breakdown} $\rightarrow$ Menguji ketatnya ($\epsilon$-$\delta$) kekontinuan seragam / keanalitikan fungsi kompleks $\rightarrow$ Rekonstruksi pembuktian teoretis agar algoritma valid kembali.
        \end{itemize}
    \end{block}
\end{frame}

% Slide 6d: Klaster Komputasi & Pemrograman (Alpro 1 \& 2, Metnum, Tek. Simulasi)
\begin{frame}{Contoh CBM – Klaster Komputasi \& Metode Numerik}
    \begin{block}{7. Algoritma \& Pemrograman Komputer 1 \& 2}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Desain sistem antrean dan ekstraksi data pasien di RS.
            \item \textbf{Alur Kasus:} Membedah kebutuhan struktur data (Array, Linked List, Sorting) $\rightarrow$ Merancang algoritma modular $\rightarrow$ Menguji \textit{time/space complexity} program.
        \end{itemize}
    \end{block}

    \begin{block}{8. Metode Numerik \& Teknik Simulasi}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Galat pencocokan posisi GPS atau Simulasi Monte Carlo risiko investasi.
            \item \textbf{Alur Kasus:} Input data acak/sinyal $\rightarrow$ Eksplorasi interpolasi numerik \& propagasi galat (\textit{truncation error}) $\rightarrow$ Menentukan metode paling efisien dan konvergen.
        \end{itemize}
    \end{block}
\end{frame}

% Slide 6e: Klaster Persamaan Diferensial & Pemodelan (PDB, PDP, Pemodelan, Mat. Sistem)
\begin{frame}{Contoh CBM – Klaster Persamaan Diferensial \& Pemodelan}
    \begin{block}{9. Persamaan Diferensial Biasa (PDB) \& Pemodelan Matematika}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Dinamika penyebaran penyakit menular (Model SIR) atau interaksi predator-mangsa.
            \item \textbf{Alur Kasus:} Observasi laju infeksi $\rightarrow$ Menyusun sistem PDB non-linear $\rightarrow$ Analisis titik kesetimbangan (\textit{equilibrium}) \& kestabilan $\rightarrow$ Rekomendasi kebijakan intervensi.
        \end{itemize}
    \end{block}

    \begin{block}{10. Persamaan Diferensial Parsial (PDP) \& Matematika Sistem}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Penyebaran polutan/panas pada pelat komponen mesin industri.
            \item \textbf{Alur Kasus:} Penentuan syarat batas/awal $\rightarrow$ Membentuk Persamaan Panas/Difusi $\rightarrow$ Menyelesaikan via Separasi Variabel / Deret Fourier $\rightarrow$ Interpretasi profil suhu/zat.
        \end{itemize}
    \end{block}
\end{frame}

% Slide 6f: Klaster Statistika & Riset Operasi (Metstat, Teori Peluang, Matstat, RO I)
\begin{frame}{Contoh CBM – Klaster Probabilitas, Statistika, \& Riset Operasi}
    \begin{block}{11. Teori Peluang, Metode Statistika, \& Matematika Statistika}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Pengujian klaim efektivitas vaksin baru atau analisis risiko klaim asuransi.
            \item \textbf{Alur Kasus:} Eksplorasi data sampel $\rightarrow$ Menentukan distribusi peluang \& estimator terbaik (MLE/UMVU) $\rightarrow$ Pengujian Hipotesis $\rightarrow$ Penarikan kesimpulan keputusan bisnis/medis.
        \end{itemize}
    \end{block}

    \begin{block}{12. Riset Operasi I}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Optimasi rute distribusi logistik bahan pokok nasional.
            \item \textbf{Alur Kasus:} Analisis kapasitas \& permintaan $\rightarrow$ Membentuk model Pemrograman Linier / Transportasi $\rightarrow$ Penyelesaian dengan Algoritma Simplex $\rightarrow$ Analisis sensitivitas biaya.
        \end{itemize}
    \end{block}
\end{frame}

% Slide Pilihan 1: Klaster Analisis, Geometri, & Aljabar
\begin{frame}{CBM pada MK Pilihan Analisis, Geometri, \& Aljabar}
    \begin{block}{1. Logika Formal / Pengantar Analisis Fungsional (Pilihan Sem 6)}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Anomali kekonvergenan pada kalkulasi ruang fungsi berdimensi tak hingga.
            \item \textbf{Alur Kasus:} Membedah ruang Hilbert/Banach yang meledak $\rightarrow$ Menguji ketetapan sifat ruang metrik \& kelengkapan (\textit{completeness}) $\rightarrow$ Rekonstruksi pembuktian formal.
        \end{itemize}
    \end{block}

    \begin{block}{2. Geometri Diferensial / Aljabar Fuzzy / Kalkulus Quantum (Pilihan Sem 7)}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Analisis kelengkungan permukaan (\textit{curvature}) pada pemetaan lintasan robot/LMS.
            \item \textbf{Alur Kasus:} Menggunakan Tensor \& Manifold $\rightarrow$ Menghitung geodesik terpendek di ruang melengkung $\rightarrow$ Formulasi optimasi gerakan robot.
        \end{itemize}
    \end{block}
\end{frame}

% Slide Pilihan 2: Klaster Persamaan Diferensial, Sistem, & Kontrol
\begin{frame}{CBM pada MK Pilihan PD, Sistem, \& Kontrol}
    \begin{block}{3. Pengantar Verifikasi Sistem / Pemodelan Mat. Sistem (Pilihan Sem 6)}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Kebocoran logika keselamatan pada pengontrol lalu lintas udara terdistribusi.
            \item \textbf{Alur Kasus:} Membangun model \textit{timed automata} $\rightarrow$ Formulasi spesifikasi LTL/CTL $\rightarrow$ Eksekusi verifikasi via SPIN/UPPAAL $\rightarrow$ Analisis jejak \textit{counterexample}.
        \end{itemize}
    \end{block}

    \begin{block}{4. Pengantar Sistem \& Kontrol / Estimasi Optimum / PDP Numerik (Pilihan Sem 7)}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Desain pengontrol penerbangan otomatis terhadap gangguan badai (\textit{turbulence}).
            \item \textbf{Alur Kasus:} Formulasi model ruang keadaan (\textit{state-space}) $\rightarrow$ Estimasi kondisi variabel via Filter Kalman $\rightarrow$ Simulasi kestabilan sistem kontrol.
        \end{itemize}
    \end{block}
\end{frame}

% Slide Pilihan 3: Klaster Finansial, Risiko, & Industri
\begin{frame}{CBM pada MK Pilihan Matematika Finansial \& Risiko}
    \begin{block}{5. Pengantar Matematika Derivatif / Pengantar Analisis Risiko (Pilihan Sem 6)}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Penentuan harga wajar opsi saham terintegrasi risiko fluktuasi pasar ekstrem.
            \item \textbf{Alur Kasus:} Analisis pergerakan harga historis $\rightarrow$ Membentuk persamaan Black-Scholes / Pohon Binomial $\rightarrow$ Mengukur indikator \textit{Value at Risk} (VaR) $\rightarrow$ Rekomendasi portofolio.
        \end{itemize}
    \end{block}

    \begin{block}{6. Pengantar Kalkulus Stokastik / Pengantar Mat. Keuangan (Pilihan Sem 7)}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Evaluasi risiko kebangkrutan perusahaan berbasis proses lompatan (\textit{jump-diffusion}).
            \item \textbf{Alur Kasus:} Identifikasi proses Wiener/Itô $\rightarrow$ Formulasi Persamaan Diferensial Stokastik (SDE) $\rightarrow$ Simulasi Monte Carlo $\rightarrow$ Rekomendasi batas alokasi modal.
        \end{itemize}
    \end{block}
\end{frame}

% Slide Pilihan 4: Klaster Sains Data, AI, & Komputasi
\begin{frame}{CBM pada MK Pilihan Sains Data, AI, \& Komputasi}
    \begin{block}{7. Pengantar Sains Data / Kecerdasan Tiruan / Pengolahan Citra (Pilihan Sem 6)}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Deteksi awal kelainan sel kanker pada citra medis radiologi.
            \item \textbf{Alur Kasus:} Preprocessing matriks citra $\rightarrow$ Ekstraksi fitur visual via transformasi wavelet/graf $\rightarrow$ Klasifikasi AI/Machine Learning $\rightarrow$ Evaluasi sensitivitas deteksi.
        \end{itemize}
    \end{block}

    \begin{block}{8. Pengantar Pembelajaran Mesin / Penambangan Data / Kriptografi (Pilihan Sem 7)}
        \begin{itemize}
            \item \textbf{Skenario Kasus:} Deteksi transaksi penipuan perbankan \textit{credit card fraud} secara \textit{real-time}.
            \item \textbf{Alur Kasus:} Penanganan \textit{imbalanced dataset} $\rightarrow$ Pelatihan model \textit{Ensemble Learning} / Neural Network $\rightarrow$ Pengujian ketahanan enkripsi data $\rightarrow$ Evaluasi metrik F1-Score.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Merancang Asesmen \& Rubrik Penilaian}
    \textbf{Prinsip Asesmen CBM:}\\ 
    Evaluasi tidak hanya berfokus pada hasil akhir (produk), tetapi sangat menghargai alur berpikir (proses).
    
    \vspace{0.4cm}
    \textbf{Rekomendasi Komponen Evaluasi (Memenuhi IKU 7):}
    \begin{itemize}
        \item \textbf{15\%} - Partisipasi Diskusi \& Kerja Kelompok (\textit{Soft skills})
        \item \textbf{25\%} - Formulasi \& Pembentukan Model Matematika
        \item \textbf{25\%} - Eksekusi, Komputasi, \& Solusi Teknis
        \item \textbf{35\%} - Laporan Akhir, Interpretasi Hasil, \& Presentasi
    \end{itemize}
    \vspace{0.2cm}
    \textit{*Total porsi berbasis kasus wajib $\ge 50\%$ dari nilai akhir MK.}
\end{frame}
